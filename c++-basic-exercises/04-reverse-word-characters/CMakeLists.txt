cmake_minimum_required(VERSION 3.20)
project(reverse_word_characters LANGUAGES CXX)
include(CTest)
add_executable(reverse_word_characters_tests tests.cpp)
target_compile_features(reverse_word_characters_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(reverse_word_characters_tests PRIVATE /W4 /WX)
else()
    target_compile_options(reverse_word_characters_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME reverse_word_characters_tests COMMAND reverse_word_characters_tests)
