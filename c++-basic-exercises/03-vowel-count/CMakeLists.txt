cmake_minimum_required(VERSION 3.20)
project(vowel_count LANGUAGES CXX)
include(CTest)
add_executable(vowel_count_tests tests.cpp)
target_compile_features(vowel_count_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(vowel_count_tests PRIVATE /W4 /WX)
else()
    target_compile_options(vowel_count_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME vowel_count_tests COMMAND vowel_count_tests)
